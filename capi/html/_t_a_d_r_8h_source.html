131void evaluateCoefficients(const int rowptr[nSpace],
132 const int colind[nnz],
133 const double v[nSpace],
134 const double alpha_L,
135 const double alpha_T,
136 const double Dm,
137 const double thetaW,
138 const double rho_f,
139 const double rho_s,
140 const double& u,
141 double& rho_out,
142 double& m,
143 double& dm,
144 double f[nSpace],
145 double df[nSpace],
146 double a[nnz],
147 double da[nnz])
148 {
149 const double epsilon = (rho_s - rho_f)/rho_f;
150 rho_out = rho_f*(1.0 + epsilon*u);
151 const double drho_du = rho_f*epsilon;
152 const double rho_transport = rho_out;
153 m = thetaW*rho_transport*u;
154 dm = thetaW*(rho_transport + u*drho_du);
155 double v_mag = 0.0;
156 for (int I=0; I<nSpace; I++)
157 v_mag += v[I]*v[I];
158 v_mag = std::sqrt(v_mag);
159 const double v_pore_mag = (thetaW > 1.0e-8) ? v_mag / thetaW : 0.0;
160
161 double alpha_L_eff = alpha_L;
162 double alpha_T_eff = alpha_T;
163 double v_unit[nSpace] = {0.0};
164 if (v_mag > 1.0e-10)
165 for (int I=0; I<nSpace; I++)
166 v_unit[I] = v[I]/v_mag;
167
168 for (int I=0; I < nSpace; I++)
169 {
170 f[I] = rho_transport*v[I]*u;
171 df[I] = (rho_transport + u*drho_du)*v[I];
172 for (int ii = rowptr[I]; ii < rowptr[I + 1]; ii++)
173 {
174 const int J = colind[ii];
175 const double deltaIJ = (I == J) ? 1.0 : 0.0;
176 const double dispersion_tensor =
177 Dm*deltaIJ +
178 alpha_L_eff*v_unit[I]*v_unit[J]*v_pore_mag +
179 alpha_T_eff*v_pore_mag*(deltaIJ - v_unit[I]*v_unit[J]);
180 a[ii] = thetaW*rho_transport*dispersion_tensor;
181 da[ii] = thetaW*drho_du*dispersion_tensor;
182 }
183
184 }

186inline
187double inversevaluateCoefficients(const double storage,
188 const double porosity,
189 const double rho_f,
190 const double rho_s)
191 {
192 const double mass_scale = std::max(porosity*rho_f, 1.0e-14);
193 const double epsilon = (rho_s - rho_f)/rho_f;
194 const double rhs = storage/mass_scale;
195 if (std::fabs(epsilon) < 1.0e-14)
196 return rhs;
197 const double discriminant = std::max(1.0 + 4.0*epsilon*rhs, 0.0);
198 const double sqrt_discriminant = std::sqrt(discriminant);
199 if (epsilon > 0.0)
200 return 2.0*rhs/(1.0 + sqrt_discriminant);
201 return (-1.0 + sqrt_discriminant)/(2.0*epsilon);
202 }
