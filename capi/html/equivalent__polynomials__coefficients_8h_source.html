2853 // Sutherland-Hodgman: keep the vertices on the wanted side of the affine functional s,
2854 // inserting the crossing point on every edge that changes side.
2855 template <class S>
2856 inline Poly clip(const Poly &in, const S &s, bool keep_positive)
2857 {
2858 Poly out;
2859 for (int i = 0; i < in.n; ++i)
2860 {
2861 const int j = (i + 1) % in.n;
2862 const double si = s(in.x[i], in.y[i]);
2863 const double sj = s(in.x[j], in.y[j]);
2864 const bool ini = keep_positive ? (si >= 0.0) : (si <= 0.0);
2865 const bool inj = keep_positive ? (sj >= 0.0) : (sj <= 0.0);
2866 if (ini) out.push(in.x[i], in.y[i]);
2867 if (ini != inj)
2868 {
2869 const double d = si - sj;
2870 const double t = (std::fabs(d) > 0.0) ? si / d : 0.0;
2871 out.push(in.x[i] + t * (in.x[j] - in.x[i]),
2872 in.y[i] + t * (in.y[j] - in.y[i]));
2873 }
2874 }
2875 if (out.n < 3) out.n = 0; // degenerate: zero measure
2876 return out;
2877 }

2900 for (int j = 0; j <= nP - i; ++j, ++idx)
2901 {

2903 for (int q = 0; q < i; ++q) t *= x;
2904 for (int q = 0; q < j; ++q) t *= y;

2919 template <int nP>
2920 inline void triangle_moments(const double p0[2], const double p1[2], const double p2[2],
2921 double *moments)
2922 {
2923 const int nDOF = ((nP + 1) * (nP + 2)) / 2;
2924 const double J00 = p1[0] - p0[0], J01 = p2[0] - p0[0];
2925 const double J10 = p1[1] - p0[1], J11 = p2[1] - p0[1];
2926 const double detJ = std::fabs(J00 * J11 - J01 * J10);
2927 if (detJ == 0.0) return;
2928 const double *gp = gl_nodes(), *gw = gl_weights();
2929 double m[((nP + 1) * (nP + 2)) / 2];
2930 for (int a = 0; a < 6; ++a)
2931 for (int b = 0; b < 6; ++b)
2932 {
2933 const double u = gp[a], v = gp[b]; // Duffy map, jacobian (1-u)
2934 const double xi = u, eta = (1.0 - u) * v;
2935 const double w = gw[a] * gw[b] * (1.0 - u) * detJ;
2936 monomials<nP>(p0[0] + J00 * xi + J01 * eta,
2937 p0[1] + J10 * xi + J11 * eta, m);
2938 for (int i = 0; i < nDOF; ++i) moments[i] += w * m[i];
2939 }
2940 }

2979 // Composite Dirac, general form: fit the Dirac measure of ONE level set's own zero curve,
2980 // {phi_target = 0}, split by the SIGN of a second, independent level set phi_split.
2981 //
2982 // Any Nitsche/Dirac-delta term that integrates over a curve inside an element cut by a
2983 // SECOND, independent curve needs this: weighting a single-cut Dirac by a region fit
2984 // (e.g. ImH_f * D_s) is a product-of-two-fits error, not moment-exact -- measured at ~31%
2985 // on the segment length, worse than the volume-weight case (see composite::weights above).
2986 // So fit one Dirac per sub-segment instead, exactly as for the volume regions: right-hand
2987 // side = the monomial moments along that piece of {phi_target=0}, then multiply by the
2988 // inverse Gram matrix.
2989 //
2990 // nodes : the element's three physical vertices, (x,y) pairs. Needed only for the scale
2991 // factor below.
2992 // cLeft : Dirac fit for {phi_target = 0} inside {phi_split < 0}
2993 // cRight : Dirac fit for {phi_target = 0} inside {phi_split > 0}
2994 //
2995 // Two current uses, both instances of this one general fit:
2996 // - dirac_weights(phi_s, phi_f, ...) below: Gamma_s's own Dirac measure, split by which
2997 // fluid phase borders each stretch (cLeft = water, cRight = air) -- the Nitsche term
2998 // on Gamma_s needs both, since the water and air sides carry different material data.
2999 // - Gamma_f's own Dirac measure (e.g. the prescribed-flux-jump source, [beta du/dn]*D_f
3000 // in ADR.h), split by edge_weights(phi_f, phi_s, ..., cMasked, cActive): the piece
3001 // inside the solid (cLeft = cMasked) must NOT contribute -- only cRight = cActive (the
3002 // part of Gamma_f actually still inside the active fluid domain) belongs in the
3003 // residual. Using the plain, single-curve Dirac fit for Gamma_f here (ignoring Gamma_s
3004 // entirely) double-counts whatever part of Gamma_f's chord falls inside the masked-out
3005 // solid -- exactly the composite-vs-product error above, for the flux-jump term
3006 // specifically.
3007 //
3008 // Scale: the moments are built in reference arc length, but the assembly integrates with the
3009 // physical volume measure, so each fit carries kappa = L_phys / (L_ref * |J|). The map is
3010 // affine and the segment straight, so kappa is a single constant.
3011 template <int nP>
3012 inline bool edge_weights(const double phi_target[3], const double phi_split[3], const double nodes[6],
3013 double *cLeft, double *cRight)
3014 {
3015 const int nDOF = ((nP + 1) * (nP + 2)) / 2;
3016 // endpoints of {phi_target = 0} on the reference triangle
3017 const double V[3][2] = {{0., 0.}, {1., 0.}, {0., 1.}};
3018 double Q[2][2];
3019 int nq = 0;
3020 for (int e = 0; e < 3 && nq < 2; ++e)
3021 {
3022 const int a = e, b = (e + 1) % 3;
3023 const double sa = phi_target[a], sb = phi_target[b];
3024 if (sa * sb < 0.0)
3025 {
3026 const double t = sa / (sa - sb);
3027 Q[nq][0] = V[a][0] + t * (V[b][0] - V[a][0]);
3028 Q[nq][1] = V[a][1] + t * (V[b][1] - V[a][1]);
3029 ++nq;
3030 }
3031 }
3032 if (nq != 2) return false;
3033
3034 const double Lref = std::sqrt((Q[1][0] - Q[0][0]) * (Q[1][0] - Q[0][0]) +
3035 (Q[1][1] - Q[0][1]) * (Q[1][1] - Q[0][1]));
3036 if (Lref <= 0.0) return false;
3037 // affine map reference -> physical
3038 const double J00 = nodes[2] - nodes[0], J01 = nodes[4] - nodes[0];
3039 const double J10 = nodes[3] - nodes[1], J11 = nodes[5] - nodes[1];
3040 const double detJ = std::fabs(J00 * J11 - J01 * J10);
3041 if (detJ <= 0.0) return false;
3042 const double dx = J00 * (Q[1][0] - Q[0][0]) + J01 * (Q[1][1] - Q[0][1]);
3043 const double dy = J10 * (Q[1][0] - Q[0][0]) + J11 * (Q[1][1] - Q[0][1]);
3044 const double Lphys = std::sqrt(dx * dx + dy * dy);
3045 const double kappa = Lphys / (Lref * detJ);
3046
